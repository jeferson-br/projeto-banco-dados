SET NAMES utf8mb4;
CREATE DATABASE automoto_pecas_central
  CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci;
USE automoto_pecas_central;

CREATE TABLE cliente (
  id_cliente  INT AUTO_INCREMENT PRIMARY KEY,
  nome        VARCHAR(100) NOT NULL,
  cpf         CHAR(14)     NOT NULL UNIQUE,
  telefone    VARCHAR(20)  NOT NULL,
  email       VARCHAR(100) UNIQUE,
  cidade      VARCHAR(60)  NOT NULL,
  uf          CHAR(2)      NOT NULL
);

CREATE TABLE funcionario (
  id_funcionario INT AUTO_INCREMENT PRIMARY KEY,
  nome           VARCHAR(100) NOT NULL,
  cpf            CHAR(14)     NOT NULL UNIQUE,
  cargo          VARCHAR(40)  NOT NULL,
  data_admissao  DATE         NOT NULL
);

CREATE TABLE fornecedor (
  id_fornecedor INT AUTO_INCREMENT PRIMARY KEY,
  razao_social  VARCHAR(120) NOT NULL,
  cnpj          CHAR(18)     NOT NULL UNIQUE,
  telefone      VARCHAR(20)  NOT NULL,
  email         VARCHAR(100) NOT NULL UNIQUE,
  cidade        VARCHAR(60)  NOT NULL,
  uf            CHAR(2)      NOT NULL
);

CREATE TABLE categoria (
  id_categoria INT AUTO_INCREMENT PRIMARY KEY,
  nome         VARCHAR(60)  NOT NULL UNIQUE,
  descricao    VARCHAR(200)
);

CREATE TABLE produto (
  id_produto     INT AUTO_INCREMENT PRIMARY KEY,
  id_categoria   INT           NOT NULL,
  id_fornecedor  INT           NOT NULL,
  nome           VARCHAR(120)  NOT NULL UNIQUE,
  tipo_veiculo   VARCHAR(5)    NOT NULL,
  preco          DECIMAL(10,2) NOT NULL,
  estoque        INT           NOT NULL DEFAULT 0,
  estoque_minimo INT           NOT NULL DEFAULT 5,
  CONSTRAINT fk_produto_categoria FOREIGN KEY (id_categoria)
    REFERENCES categoria (id_categoria),
  CONSTRAINT fk_produto_fornecedor FOREIGN KEY (id_fornecedor)
    REFERENCES fornecedor (id_fornecedor),
  CONSTRAINT ck_produto_tipo CHECK (tipo_veiculo IN ('Carro','Moto','Ambos')),
  CONSTRAINT ck_produto_preco CHECK (preco > 0),
  CONSTRAINT ck_produto_estoque CHECK (estoque >= 0 AND estoque_minimo >= 0)
);

CREATE TABLE venda (
  id_venda       INT AUTO_INCREMENT PRIMARY KEY,
  id_cliente     INT      NOT NULL,
  id_funcionario INT      NOT NULL,
  data_venda     DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT fk_venda_cliente FOREIGN KEY (id_cliente)
    REFERENCES cliente (id_cliente),
  CONSTRAINT fk_venda_funcionario FOREIGN KEY (id_funcionario)
    REFERENCES funcionario (id_funcionario)
);

CREATE TABLE item_venda (
  id_venda       INT           NOT NULL,
  id_produto     INT           NOT NULL,
  quantidade     INT           NOT NULL,
  preco_unitario DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (id_venda, id_produto),
  CONSTRAINT fk_item_venda FOREIGN KEY (id_venda)
    REFERENCES venda (id_venda),
  CONSTRAINT fk_item_produto FOREIGN KEY (id_produto)
    REFERENCES produto (id_produto),
  CONSTRAINT ck_item_quantidade CHECK (quantidade > 0),
  CONSTRAINT ck_item_preco CHECK (preco_unitario > 0)
);

CREATE TABLE pagamento (
  id_pagamento    INT AUTO_INCREMENT PRIMARY KEY,
  id_venda        INT           NOT NULL,
  forma_pagamento VARCHAR(20)   NOT NULL,
  valor           DECIMAL(10,2) NOT NULL,
  parcelas        INT           NOT NULL DEFAULT 1,
  data_pagamento  DATETIME,
  status          VARCHAR(10)   NOT NULL,
  CONSTRAINT fk_pagamento_venda FOREIGN KEY (id_venda)
    REFERENCES venda (id_venda),
  CONSTRAINT ck_pag_forma CHECK (forma_pagamento IN
    ('Dinheiro','PIX','Cartão de débito','Cartão de crédito')),
  CONSTRAINT ck_pag_valor CHECK (valor > 0),
  CONSTRAINT ck_pag_parcelas CHECK (parcelas BETWEEN 1 AND 12
    AND (forma_pagamento = 'Cartão de crédito' OR parcelas = 1)),
  CONSTRAINT ck_pag_status CHECK (
    (status = 'Pago' AND data_pagamento IS NOT NULL) OR
    (status = 'Pendente' AND data_pagamento IS NULL))
);

CREATE VIEW vw_vendas_detalhadas AS
SELECT v.id_venda,
       v.data_venda,
       c.nome AS cliente,
       f.nome AS funcionario,
       p.nome AS produto,
       i.quantidade,
       i.preco_unitario,
       i.quantidade * i.preco_unitario AS subtotal
FROM venda v
INNER JOIN cliente     c ON c.id_cliente     = v.id_cliente
INNER JOIN funcionario f ON f.id_funcionario = v.id_funcionario
INNER JOIN item_venda  i ON i.id_venda       = v.id_venda
INNER JOIN produto     p ON p.id_produto     = i.id_produto;

CREATE VIEW vw_estoque AS
SELECT p.nome AS produto,
       c.nome AS categoria,
       p.tipo_veiculo,
       p.preco,
       p.estoque,
       CASE
         WHEN p.estoque = 0 THEN 'Sem estoque'
         WHEN p.estoque <= p.estoque_minimo THEN 'Estoque baixo'
         ELSE 'Estoque normal'
       END AS situacao_estoque
FROM produto p
INNER JOIN categoria c ON c.id_categoria = p.id_categoria;

SELECT id_produto, nome, tipo_veiculo, preco, estoque
FROM produto;

SELECT nome, tipo_veiculo, preco
FROM produto
WHERE tipo_veiculo IN ('Carro','Ambos');

SELECT nome, tipo_veiculo, preco
FROM produto
WHERE tipo_veiculo IN ('Moto','Ambos');

SELECT p.nome, p.preco
FROM produto p
INNER JOIN categoria c ON c.id_categoria = p.id_categoria
WHERE c.nome = 'Freios';

SELECT nome, estoque
FROM produto
WHERE estoque < 5
ORDER BY estoque;

SELECT DISTINCT c.id_cliente, c.nome
FROM cliente c
INNER JOIN venda v ON v.id_cliente = c.id_cliente
ORDER BY c.nome;

SELECT v.id_venda, v.data_venda, c.nome AS cliente, f.nome AS funcionario
FROM venda v
INNER JOIN cliente c ON c.id_cliente = v.id_cliente
INNER JOIN funcionario f ON f.id_funcionario = v.id_funcionario
ORDER BY v.data_venda;

SELECT i.id_venda, p.nome AS produto, i.quantidade, i.preco_unitario
FROM item_venda i
INNER JOIN produto p ON p.id_produto = i.id_produto
ORDER BY i.id_venda, p.nome;

SELECT id_venda, SUM(quantidade * preco_unitario) AS valor_total
FROM item_venda
GROUP BY id_venda
ORDER BY id_venda;

SELECT COUNT(DISTINCT id_venda) AS numero_de_vendas,
       SUM(quantidade * preco_unitario) AS valor_vendido
FROM item_venda;

SELECT c.id_cliente, c.nome,
       COALESCE(SUM(i.quantidade * i.preco_unitario), 0) AS total_gasto
FROM cliente c
LEFT JOIN venda v ON v.id_cliente = c.id_cliente
LEFT JOIN item_venda i ON i.id_venda = v.id_venda
GROUP BY c.id_cliente, c.nome
ORDER BY total_gasto DESC;

SELECT nome, preco
FROM produto
WHERE preco = (SELECT MAX(preco) FROM produto);

SELECT nome, preco
FROM produto
WHERE preco = (SELECT MIN(preco) FROM produto);

SELECT ROUND(AVG(preco), 2) AS preco_medio
FROM produto;

SELECT c.nome AS categoria, COUNT(p.id_produto) AS total_produtos
FROM categoria c
LEFT JOIN produto p ON p.id_categoria = c.id_categoria
GROUP BY c.id_categoria, c.nome
ORDER BY total_produtos DESC, c.nome;

SELECT id_venda, data_venda, id_cliente
FROM venda
WHERE data_venda BETWEEN '2026-10-01 00:00:00' AND '2026-10-31 23:59:59'
ORDER BY data_venda;

SELECT nome, preco
FROM produto
WHERE nome LIKE '%filtro%';

SELECT p.nome, c.nome AS categoria
FROM produto p
INNER JOIN categoria c ON c.id_categoria = p.id_categoria
WHERE c.nome IN ('Pneus','Lubrificantes')
ORDER BY c.nome, p.nome;

SELECT nome, preco, estoque
FROM produto
WHERE tipo_veiculo = 'Moto'
  AND (preco < 50.00 OR estoque <= estoque_minimo);

SELECT nome, preco, estoque
FROM produto
ORDER BY preco DESC, estoque ASC;

SELECT * FROM vw_vendas_detalhadas ORDER BY id_venda, produto;
SELECT * FROM vw_estoque ORDER BY situacao_estoque, produto;

SELECT t.id_venda, t.total_itens, pg.total_pagamentos
FROM (SELECT id_venda, SUM(quantidade * preco_unitario) AS total_itens
      FROM item_venda GROUP BY id_venda) t
LEFT JOIN (SELECT id_venda, SUM(valor) AS total_pagamentos
           FROM pagamento GROUP BY id_venda) pg ON pg.id_venda = t.id_venda
WHERE pg.total_pagamentos IS NULL OR pg.total_pagamentos <> t.total_itens;
